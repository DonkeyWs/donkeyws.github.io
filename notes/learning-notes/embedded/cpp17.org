** 楔子(Prologue)

~clang~ 和现代 ~C++~ 正在逐步成为嵌入式开发的可用选择方案，本文尝试通过现代 ~C++~ （ ~C++17~ ）实现个简单的嵌入式项目，
以了解使用现代 ~C++~ 开发嵌入式项目的方法步骤和注意事项，并从中探索在嵌入式项目中使用 ~C++~ 的优劣势。

** 环境准备(Setup)

** STM32F103

- CPU：Cortex-M3
- Thumb 指令集
- 无 ~FPU~ ，通常使用 ~soft-float~

对于 ~clang++~ 来说，核心编译参数如下：

  --target=thumbv7m-none-eabi -mcpu=cortex-m3 -mthumb -mfloat-abi=soft -std=c++17 -Os -g3 -ffunction-sections -fdata-sections -fno-common -ffreestanding -fno-builtin -fno-exceptions -fno-rtti -fno-threadsafe-statics -fno-use-cxa-atexit -Wall -Wextra -fno-strict-aliasing -fno-stack-protector

~linker/stm32f103c8tx.ld~:

#+begin_src ldscript
  ENTRY(Reset_Handler)

  MEMORY
  {
      FLASH (rx)  : ORIGIN = 0x08000000, LENGTH = 64K
      RAM   (rwx) : ORIGIN = 0x20000000, LENGTH = 20K
  }

  _estack = ORIGIN(RAM) + LENGTH(RAM);

  _Min_Heap_Size  = 0x200;
  _Min_Stack_Size = 0x400;

  SECTIONS
  {
      /* 中断向量表必须放在最前面 */
      .isr_vector :
      {
          . = ALIGN(4);
          KEEP(*(.isr_vector))
          . = ALIGN(4);
      } >FLASH

      .text :
      {
          . = ALIGN(4);

          ,*(.text .text.* .gnu.linkonce.t.*)
          ,*(.glue_7t) *(.glue_7)

          KEEP(*(.init))
          KEEP(*(.fini))

          . = ALIGN(4);
          _etext = .;
      } >FLASH

      .rodata :
      {
          . = ALIGN(4);
          ,*(.rodata .rodata.* .gnu.linkonce.r.*)
          . = ALIGN(4);
      } >FLASH

      /* C++ 异常/展开表，即使禁用异常也建议保留 */
      .ARM.extab :
      {
          ,*(.ARM.extab* .gnu.linkonce.armextab.*)
      } >FLASH

      .ARM.exidx :
      {
          __exidx_start = .;
          ,*(.ARM.exidx* .gnu.linkonce.armexidx.*)
          __exidx_end = .;
      } >FLASH

      /* C/C++ 初始化函数表 */
      .preinit_array :
      {
          PROVIDE_HIDDEN(__preinit_array_start = .);
          KEEP(*(.preinit_array*))
          PROVIDE_HIDDEN(__preinit_array_end = .);
      } >FLASH

      .init_array :
      {
          PROVIDE_HIDDEN(__init_array_start = .);
          KEEP(*(SORT_BY_INIT_PRIORITY(.init_array.*)))
          KEEP(*(.init_array*))
          PROVIDE_HIDDEN(__init_array_end = .);
      } >FLASH

      .fini_array :
      {
          PROVIDE_HIDDEN(__fini_array_start = .);
          KEEP(*(SORT_BY_INIT_PRIORITY(.fini_array.*)))
          KEEP(*(.fini_array*))
          PROVIDE_HIDDEN(__fini_array_end = .);
      } >FLASH

      /* .data 段：运行地址在 RAM，加载地址在 Flash */
      _sidata = LOADADDR(.data);

      .data :
      {
          . = ALIGN(4);
          _sdata = .;

          ,*(.data .data.* .gnu.linkonce.d.*)
          ,*(.data.rel.ro .data.rel.ro.*)
          ,*(.ramfunc .ramfunc.*)

          . = ALIGN(4);
          _edata = .;
      } >RAM AT>FLASH

      .bss :
      {
          . = ALIGN(4);
          _sbss = .;
          __bss_start__ = _sbss;

          ,*(.bss .bss.* .gnu.linkonce.b.*)
          ,*(COMMON)

          . = ALIGN(4);
          _ebss = .;
          __bss_end__ = _ebss;
      } >RAM

      /* 简单 heap/stack 占位 */
      ._user_heap_stack (NOLOAD) :
      {
          . = ALIGN(8);
          PROVIDE(end = .);
          PROVIDE(_end = .);

          . = . + _Min_Heap_Size;
          . = . + _Min_Stack_Size;

          . = ALIGN(8);
      } >RAM
  }
#+end_src

~src/main.cpp~:

#+begin_src cpp
  #include <stdint.h>

  namespace hw {

  template <uint32_t Addr>
  inline volatile uint32_t& reg() noexcept {
      return *reinterpret_cast<volatile uint32_t*>(Addr);
  }

  } // namespace hw

  namespace board {

  using hw::reg;

  // RCC
  inline constexpr uint32_t rcc_base       = 0x40021000;
  inline constexpr uint32_t rcc_apb2enr    = rcc_base + 0x18;
  inline constexpr uint32_t rcc_apb2enr_iopc = 1u << 4;

  // GPIOC
  inline constexpr uint32_t gpioc_base = 0x40011000;
  inline constexpr uint32_t gpioc_crh  = gpioc_base + 0x04;
  inline constexpr uint32_t gpioc_odr  = gpioc_base + 0x0c;

  // Blue Pill LED: PC13, active low
  inline constexpr uint32_t led_pin = 13;
  inline constexpr bool led_active_low = true;

  // C++17: static_assert 可以没有 message
  static_assert(led_pin == 13);

  inline void led_set(bool on) noexcept {
      bool level = on;

      // C++17 if constexpr
      if constexpr (led_active_low) {
          level = !level;
      }

      if (level) {
          reg<gpioc_odr>() = reg<gpioc_odr>() | (1u << led_pin);
      } else {
          reg<gpioc_odr>() = reg<gpioc_odr>() & ~(1u << led_pin);
      }
  }

  inline void led_init() noexcept {
      // 打开 GPIOC 时钟
      reg<rcc_apb2enr>() = reg<rcc_apb2enr>() | rcc_apb2enr_iopc;

      // 配置 PC13 为推挽输出，2 MHz
      // CRH 中 pin13 对应 bit[23:20]
      uint32_t crh = reg<gpioc_crh>();
      crh &= ~(0xFu << 20);
      crh |=  (0x2u << 20);
      reg<gpioc_crh>() = crh;

      // 初始关闭 LED
      led_set(false);
  }

  } // namespace board

  struct Settings {
      uint32_t blink_delay;

      // 这个构造函数需要 C++ 运行时初始化
      Settings() noexcept : blink_delay(500000) {}
  };

  static Settings settings;

  static void delay(uint32_t n) noexcept {
      while (n--) {
          __asm volatile("nop" ::: "memory");
      }
  }

  extern "C" int main() {
      board::led_init();

      // 如果 C++ 全局构造没有生效，给一个兜底值
      uint32_t d = settings.blink_delay ? settings.blink_delay : 500000;

      while (true) {
          board::led_set(true);
          delay(d);

          board::led_set(false);
          delay(d);
      }
  }
#+end_src

~startup/startup_stm32f103.S~:

#+begin_src asm
  .syntax unified
  .cpu cortex-m3
  .thumb

  .global g_pfnVectors
  .global Default_Handler
  .global Reset_Handler

  /* disable executable stack */
  .section .note.GNU-stack,"",%progbits
  /* ========================= */
  /* 中断向量表                */
  /* ========================= */
  .section .isr_vector,"a",%progbits
  .type g_pfnVectors, %object
g_pfnVectors:
  .word _estack
  .word Reset_Handler
  .word NMI_Handler
  .word HardFault_Handler
  .word MemManage_Handler
  .word BusFault_Handler
  .word UsageFault_Handler
  .word 0
  .word 0
  .word 0
  .word 0
  .word SVC_Handler
  .word DebugMon_Handler
  .word 0
  .word PendSV_Handler
  .word SysTick_Handler
  .size g_pfnVectors, .-g_pfnVectors

  /* ========================= */
  /* Reset Handler             */
  /* ========================= */
  .section .text
  .align 2

  .thumb_func
  .type Reset_Handler, %function
Reset_Handler:
  ldr r0, =_estack
  mov sp, r0

  /* 初始化期间可先关中断 */
  cpsid i

  /* 拷贝 .data 段：从 flash 到 ram */
  ldr r0, =_sdata
  ldr r1, =_edata
  ldr r2, =_sidata
  movs r3, #0
  b 2f
1:
  ldr r4, [r2, r3]
  str r4, [r0, r3]
  adds r3, r3, #4
2:
  adds r4, r0, r3
  cmp r4, r1
  bcc 1b

  /* 清零 .bss 段 */
  ldr r0, =_sbss
  ldr r1, =_ebss
  movs r2, #0
  b 4f
3:
  str r2, [r0]
  adds r0, r0, #4
4:
  cmp r0, r1
  bcc 3b

  /* 调用 C/C++ 运行时初始化 */
  bl __libc_init_array

  /* 进入 main 前打开中断 */
  cpsie i

  bl main
  b .
  .size Reset_Handler, .-Reset_Handler

  /* ========================= */
  /* Default Handler           */
  /* ========================= */
  .thumb_func
  .type Default_Handler, %function
Default_Handler:
  b .
  .size Default_Handler, .-Default_Handler

  /* 提供空的 _init/_fini，避免 -nostartfiles 下缺少符号 */
  .weak _init
  .thumb_func
  .type _init, %function
_init:
  bx lr
  .size _init, .-_init

  .weak _fini
  .thumb_func
  .type _fini, %function
_fini:
  bx lr
  .size _fini, .-_fini

  /* ========================= */
  /* 弱别名到 Default_Handler  */
  /* ========================= */
  .weak NMI_Handler
  .thumb_set NMI_Handler, Default_Handler

  .weak HardFault_Handler
  .thumb_set HardFault_Handler, Default_Handler

  .weak MemManage_Handler
  .thumb_set MemManage_Handler, Default_Handler

  .weak BusFault_Handler
  .thumb_set BusFault_Handler, Default_Handler

  .weak UsageFault_Handler
  .thumb_set UsageFault_Handler, Default_Handler

  .weak SVC_Handler
  .thumb_set SVC_Handler, Default_Handler

  .weak DebugMon_Handler
  .thumb_set DebugMon_Handler, Default_Handler

  .weak PendSV_Handler
  .thumb_set PendSV_Handler, Default_Handler

  .weak SysTick_Handler
  .thumb_set SysTick_Handler, Default_Handler
#+end_src

使用 =renode= 来模拟 =STM32F103= 这个板子，方便调试，首先创建 =RESC= 脚本 =run_stm32.resc= ，内容如下：

#+begin_src
  using sysbus
  mach create "stm32f103"

  # 加载STM32F103平台描述文件
  machine LoadPlatformDescription @platforms/cpus/stm32f103.repl

  # 加载编译好的固件（ELF格式）
  machine LoadELF @/path/to/your/firmware.elf

  # 打开UART2分析器用于串口输出
  showAnalyzer sysbus.usart2

  # 配置UART外设
  showAnalyzer sysbus.usart2  # 启用UART2分析器

  # 创建LED外设
  led: LED.LED
  led: LED.LED @ sysbus.gpioc 13  # 连接到PC13引脚

  # 创建按钮外设
  button: Button.Button
  button: Button.Button @ sysbus.gpioa 0  # 连接到PA0引脚

  # SPI配置
  spi: SPI.STM32F103_SPI @ sysbus.spi1
  spi: SPI.STM32F103_SPI @ sysbus.spi2

  # I2C配置
  i2c: I2C.STM32F103_I2C @ sysbus.i2c1
  i2c: I2C.STM32F103_I2C @ sysbus.i2c2

  # 启动GDB服务器（端口3333）
  machine StartGdbServer 3333 true

  # 开始仿真
  start
#+end_src

#+begin_src makefile
PROJECT     = firmware
BUILD_DIR   = build
LDSCRIPT    = linker/stm32f103c8tx.ld

CROSS       ?= arm-none-eabi-
CC          = $(CROSS)gcc
CXX         = clang++
LD          = $(CROSS)g++
OBJCOPY     = $(CROSS)objcopy
SIZE        = $(CROSS)size

# GCC/ARM 工具链使用的 CPU 选项
ARCH_GCC    = -mcpu=cortex-m3 -mthumb -mfloat-abi=soft

# Clang 使用的 target triple
ARCH_CLANG  = --target=thumbv7m-none-eabi -mcpu=cortex-m3 -mfloat-abi=soft

OPT         ?= -Os
DEBUG       ?= -g3

COMMON_FLAGS = \
  $(OPT) \
  $(DEBUG) \
  -fno-common \
  -fno-strict-aliasing \
  -fno-stack-protector \
  -ffunction-sections \
  -fdata-sections \
  -Wall \
  -Wextra

CFLAGS = \
  $(ARCH_GCC) \
  $(COMMON_FLAGS) \
  -std=c11 \
  -ffreestanding \
  -fno-builtin

# -fshort-enums: make sure use variable-size enum, keep in line with gcc behavors
CXXFLAGS = \
  $(ARCH_CLANG) \
  $(COMMON_FLAGS) \
  -std=c++17 \
  -fshort-enums \
  -nostdlib \
  -nodefaultlibs \
  -ffreestanding \
  -fno-builtin \
  -fno-exceptions \
  -fno-rtti \
  -fno-threadsafe-statics \
  -fno-unwind-tables \
  -fno-use-cxa-atexit \
  -D_LIBCPP_HAS_NO_THREADS \
  -D_LIBCPP_HAS_NO_MONOTONIC_CLOCK \
  -D_LIBCPP_HAS_NO_TIME_ZONE_DATABASE

ASFLAGS = \
  $(ARCH_GCC) \
  $(DEBUG) \
  -x assembler-with-cpp

# -Wl,-z,noexecstack: disable executable stack
LDFLAGS = \
  $(ARCH_GCC) \
  -T$(LDSCRIPT) \
  -nostartfiles \
  --specs=nano.specs \
  --specs=nosys.specs \
  -Wl,--gc-sections \
  -Wl,-Map=$(BUILD_DIR)/$(PROJECT).map \
  -Wl,--print-memory-usage \
  -Wl,-z,noexecstack

OBJS = \
  $(BUILD_DIR)/startup.o \
  $(BUILD_DIR)/main.o

ARM_LIBDIR := $(dir $(shell $(CROSS)g++ $(ARCH_GCC) -print-file-name=libc.a))
GCC_LIBDIR := $(dir $(shell $(CROSS)gcc $(ARCH_GCC) -print-libgcc-file-name))

all: $(BUILD_DIR)/$(PROJECT).elf $(BUILD_DIR)/$(PROJECT).bin $(BUILD_DIR)/$(PROJECT).hex
	$(SIZE) $(BUILD_DIR)/$(PROJECT).elf

$(BUILD_DIR):
	mkdir -p $@

clang-link: $(OBJS)
	$(CXX) $(ARCH_CLANG) $(OPT) $(DEBUG) \
	  -fuse-ld=lld \
	  -nostdlib \
	  -T$(LDSCRIPT) \
	  -Wl,--gc-sections \
	  -Wl,-Map=$(BUILD_DIR)/firmware_clang.map \
	  $(OBJS) -o $(BUILD_DIR)/firmware_clang.elf \
	  -L$(ARM_LIBDIR) -L$(GCC_LIBDIR) \
	  -Wl,--start-group \
	  -lstdc++ -lsupc++ -lc -lg -lnosys -lgcc \
	  -Wl,--end-group

$(BUILD_DIR)/main.o: src/main.cpp | $(BUILD_DIR)
	$(CXX) $(CXXFLAGS) -c $< -o $@

$(BUILD_DIR)/startup.o: startup/startup_stm32f103.S | $(BUILD_DIR)
	# $(CC) $(ASFLAGS) -c $< -o $@
	clang++ --target=thumbv7m-none-eabi -mcpu=cortex-m3 -mfloat-abi=soft \
		-x assembler-with-cpp -c $< \
		-o $@

$(BUILD_DIR)/$(PROJECT).elf: $(OBJS)
	$(LD) $(LDFLAGS) $(OBJS) -o $@

$(BUILD_DIR)/%.bin: $(BUILD_DIR)/%.elf
	$(OBJCOPY) -O binary $< $@

$(BUILD_DIR)/%.hex: $(BUILD_DIR)/%.elf
	$(OBJCOPY) -O ihex $< $@

clean:
	rm -rf $(BUILD_DIR)

flash-stlink:
	st-flash write $(BUILD_DIR)/$(PROJECT).bin 0x08000000

flash-openocd:
	openocd -f interface/stlink.cfg -f target/stm32f1x.cfg \
	  -c "program $(BUILD_DIR)/$(PROJECT).elf verify reset exit"

.PHONY: all clean flash-stlink flash-openocd
#+end_src
